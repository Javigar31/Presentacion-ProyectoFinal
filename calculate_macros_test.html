// Portado 1:1 desde api/nutrition.php — Fórmula Mifflin-St Jeor function calculate_macros(weight, height, age, gender, objective, activity) { const genderConst = gender === 'FEMALE' ? -161 : 5; const tmb = (10 * weight) + (6.25 * height) - (5 * age) + genderConst; const facs = { SEDENTARY:1.2, LIGHT:1.375, MODERATE:1.55, ACTIVE:1.725, VERY_ACTIVE:1.9 }; const tdee = tmb * (facs[activity] !== undefined ? facs[activity] : 1.55); const adjs = { LOSE_WEIGHT:-0.20, GAIN_MUSCLE:0.10 }; const adj = adjs[objective] !== undefined ? adjs[objective] : 0.00; const cal = Math.max(1200, tdee * (1 + adj)); const prot = weight * 2.0; const fat = (cal * 0.25) / 9; const carb = (cal - (prot * 4 + fat * 9)) / 4; return { calories: Math.round(cal), protein_g: Math.round(prot), carbs_g: Math.round(carb), fat_g: Math.round(fat), tmb: Math.round(tmb), tdee: Math.round(tdee), adjustment: Math.round(cal - tdee) }; }
